## Global Constraints

- No comments in code (repo convention). TypeScript strict.
- All user-facing strings via `t()`; every new key added to BOTH `public/locales/en/<ns>.json` and `public/locales/de/<ns>.json` (key trees must stay identical; typed keys come from `en` automatically).
- Every data view implements loading (`LoadingState`), empty (`EmptyState` + primary action where meaningful), error (`ErrorState` + retry) states.
- Every mutation: success or error notification (`notifySuccess` / `notifyMutationError`); 422 `{"errors":[...]}` payloads are additionally rendered per-field in the form and summarized in one notification (design §2.5).
- After every mutation: invalidate affected query keys via the central factory (`src/api/queryKeys.ts`); server state is never copied into client state.
- Numbers via `Intl.NumberFormat` parameterized with the active i18next language; relative timestamps via dayjs (`relativeTime` plugin, active locale). Never format by hand.
- Full gate per task, from `frontend/`: `npm test -- --run && npm run typecheck && npm run build`.
- Tests: vitest + Testing Library, jsdom; fetch stubbed via `stubFetch` from `src/test/fetch.ts` (locale serving is layered automatically). Component/App tests use the shared `render` helper.
- M10-c test conventions (binding, they were review-verified): Mantine `Select` → `getByRole('combobox', { name })`; `userEvent.setup()` replaces `navigator.clipboard` — stub it AFTER `setup()` via `Object.defineProperty` when a test asserts copy behavior; TanStack Form dirty tracking uses `form.Subscribe` (NOT `form.state.isDirty` directly — `form.state` is not reactive in v1).
- Mantine `ff` prop accepts only `'monospace' | 'text' | 'heading' | string` (no `'numeric'` token — use `monospace` + `tabular-nums`).
- dnd-kit testing: use `@testing-library/user-event` pointer events on `DndContext`; if a test is too brittle, test the underlying state-mutation functions instead (spec §10 risk).
- Notifications: `notifications.clean()` in `beforeEach` to prevent leakage between tests (M10-c lesson).
- i18n acceptance: `beforeAll(async () => { await i18n.loadNamespaces('<ns>'); })` before each test that uses a non-default namespace.
- If a step's expectation fails: fix forward only when small and faithful; otherwise report BLOCKED with details. All deviations go into the final report for the reviewer.

## Backend ground truth (verified 2026-08-29, all already implemented in M10-a)

- `GET /plugins` → `[{id, name, version, enabled, manifest: {...}, used_by_feed_sources: N}]`
  - `manifest.config_schema` (object, JSON Schema) and/or `manifest.data_schema`
  - `manifest.frontend`: `{menu_item: string, icon: string}` — icon is a tabler icon name
  - `manifest.extension_point` ∈ `pipeline_module` | `data` | `config` | etc.
  - `manifest.config_scope` / `data_scope` (optional): `'client'` | `'feed_source'` | `'global'` | list; missing → `'global'`
- `GET /plugins/{id}/config?client_id=&feed_source_id=` → `Record<string, any>` (current config or `{}`)
- `PUT /plugins/{id}/config?client_id=&feed_source_id=` body=`config` → updated config; 422 `{"errors":[...]}` on invalid
- `PUT /plugins/{id}/enabled` body=`{enabled: bool}` → updated plugin row
- `GET /feed-sources/{id}/pipeline` → `{"instances": [{position, plugin_id, name, configuration}]}`
- `PUT /feed-sources/{id}/pipeline` body=`{instances: [{plugin_id, name, configuration}]}` → full-replace; 422 `{"errors":[...]}` on unknown/disabled/non-pipeline plugin or invalid `configuration`
- `GET /feed-sources/{id}/export-history` → `[ExportVersionOut]` with `source: "scheduled"|"manual"|"rollback"`, `findings: {critical, warning, info}` (0 for rollback-source)
- `GET /feed-sources/{id}/export-history/{version}/diff?against=N` → `DiffOut {version, against, added: [product_id], removed: [product_id], changed: [DiffProductOut{product_id, fields: [DiffFieldOut{field, old, new}]}]}`
- `POST /feed-sources/{id}/export-history/{version}/rollback` → 204; creates new version with `source="rollback"`
- `GET /feed-sources/{id}/ingestion-runs?limit&offset` → `IngestionRunOut[]`
- `GET /feed-sources/{id}/quality-findings` → `{ingestion_run_id, counts, findings}`
- `POST /feed-sources/{id}/dry-run` body=`{limit?: int}` → dry-run results

```ts
export function useUpdatePluginEnabled() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: ({ id, enabled }: { id: string; enabled: boolean }) =>
      apiPut<PluginInfo>(`/plugins/${id}/enabled`, { enabled }),
    onSuccess: () => {
      void queryClient.invalidateQueries({ queryKey: queryKeys.plugins });
    },
  });
}

export type PluginScope = { clientId?: number; feedSourceId?: number };

function buildScopeQuery(scope?: PluginScope): string {
  if (!scope) return '';
  const params = new URLSearchParams();
  if (scope.clientId !== undefined) params.set('client_id', String(scope.clientId));
  if (scope.feedSourceId !== undefined) params.set('feed_source_id', String(scope.feedSourceId));
  const qs = params.toString();
  return qs ? `?${qs}` : '';
}

export function usePluginConfig(pluginId: string, scope?: PluginScope) {
  return useQuery({
    queryKey: queryKeys.pluginConfig(pluginId, scope),
    queryFn: () => apiGet<PluginConfigResponse>(`/plugins/${pluginId}/config${buildScopeQuery(scope)}`),
    enabled: Boolean(pluginId),
  });
}

export function useSavePluginConfig(pluginId: string, scope?: PluginScope) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (config: PluginConfigResponse) =>
      apiPut<PluginConfigResponse>(`/plugins/${pluginId}/config${buildScopeQuery(scope)}`, config),
    onSuccess: () => {
      void queryClient.invalidateQueries({ queryKey: queryKeys.pluginConfig(pluginId, scope) });
    },
  });
}
```

```tsx
import { Button, Group, Stack, Title } from '@mantine/core';
import { useMemo } from 'react';
import { useParams } from 'react-router';
import { useTranslation } from 'react-i18next';
import { usePluginConfig, useSavePluginConfig, usePlugins, type PluginScope } from '../../api/hooks';
import { JsonSchemaForm } from '../../components/JsonSchemaForm';
import { EmptyState, ErrorState, LoadingState } from '../../components/StateViews';
import { notifyMutationError, notifySuccess } from '../../app/notifications';
import { ApiError } from '../../api/client';

export function PluginPage() {
  const { t } = useTranslation('plugins');
  const { pluginId, clientId, feedSourceId } = useParams();
  const { data: plugins, isPending, isError, refetch } = usePlugins();

  const scope: PluginScope = useMemo(() => {
    const s: PluginScope = {};
    if (clientId) s.clientId = Number(clientId);
    if (feedSourceId) s.feedSourceId = Number(feedSourceId);
    return s;
  }, [clientId, feedSourceId]);

  const config = usePluginConfig(pluginId ?? '', scope);
  const saveConfig = useSavePluginConfig(pluginId ?? '', scope);

  if (isPending) return <LoadingState />;
  if (isError) return <ErrorState onRetry={() => void refetch()} />;

  const plugin = (plugins ?? []).find((p) => p.id === pluginId);
  if (!plugin) return <EmptyState>{t('notFound')}</EmptyState>;

  const schema = plugin.manifest.config_schema ?? plugin.manifest.data_schema;
  if (!schema) return <EmptyState>{t('noSchema')}</EmptyState>;

  async function onSubmit(value: unknown) {
    if (!pluginId) return;
    try {
      await saveConfig.mutateAsync((value ?? {}) as Record<string, unknown>);
      notifySuccess(t('configSaved'));
    } catch (error) {
      if (error instanceof ApiError && error.errors) {
        notifyMutationError(error, t('saveFailed'));
        return;
      }
      notifyMutationError(error, t('saveFailed'));
    }
  }

  return (
    <Stack gap="md">
      <Title order={3}>{plugin.name}</Title>
      {config.isPending ? (
        <LoadingState />
      ) : config.isError ? (
        <ErrorState onRetry={() => void config.refetch()} />
      ) : (
        <JsonSchemaForm
          schema={schema as Parameters<typeof JsonSchemaForm>[0]['schema']}
          value={config.data ?? {}}
          onChange={(next) => {
            // Optimistic local state via the query cache
            void next;
          }}
          errors={saveConfig.error instanceof ApiError ? mapErrors(saveConfig.error.errors) : {}}
        />
      )}
      <Group justify="flex-end">
        <Button onClick={() => void onSubmit(config.data ?? {})} loading={saveConfig.isPending}>
          {t('save')}
        </Button>
      </Group>
    </Stack>
  );
}

function mapErrors(errors: string[] | null): Record<string, string> {
  if (!errors) return {};
  const out: Record<string, string> = {};
  for (const e of errors) {
    const idx = e.indexOf(':');
    if (idx > 0) {
      out[e.slice(0, idx).trim()] = e.slice(idx + 1).trim();
    } else {
      out._form = e;
    }
  }
  return out;
}
```

```ts
import type { PipelineInstance } from '../../api/types';

export type LocalInstance = PipelineInstance & { clientId: string };

export function addInstance(
  instances: LocalInstance[],
  plugin: { id: string; name: string },
): LocalInstance[] {
  const clientId = generateId();
  return [
    ...instances,
    {
      clientId,
      plugin_id: plugin.id,
      name: plugin.name,
      configuration: {},
    },
  ];
}

export function reorderInstances(
  instances: LocalInstance[],
  fromIdx: number,
  toIdx: number,
): LocalInstance[] {
  if (fromIdx === toIdx) return instances;
  const next = instances.slice();
  const [moved] = next.splice(fromIdx, 1);
  next.splice(toIdx, 0, moved);
  return next;
}

export function removeInstance(instances: LocalInstance[], clientId: string): LocalInstance[] {
  return instances.filter((i) => i.clientId !== clientId);
}

export function isInstancesEqual(
  a: LocalInstance[] | PipelineInstance[],
  b: LocalInstance[] | PipelineInstance[],
): boolean {
  return JSON.stringify(stripClientIds(a)) === JSON.stringify(stripClientIds(b));
}

function stripClientIds(items: Array<PipelineInstance & { clientId?: string }>): PipelineInstance[] {
  return items.map(({ clientId: _clientId, ...rest }) => rest);
}

function generateId(): string {
  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) {
    return crypto.randomUUID();
  }
  return `id-${Math.random().toString(36).slice(2)}-${Date.now()}`;
}
```

```ts
export function useExportVersionDiff(
  feedSourceId: number | string,
  version: number | undefined,
  against: number | undefined,
) {
  return useQuery({
    queryKey: queryKeys.feedSource(feedSourceId).exportDiff({ version: version ?? -1, against: against ?? -1 }),
    queryFn: () => {
      const qs = against !== undefined ? `?against=${against}` : '';
      return apiGet<DiffOut>(`/feed-sources/${feedSourceId}/export-history/${version}/diff${qs}`);
    },
    enabled: version !== undefined && against !== undefined,
  });
}

export function useRollbackToVersion(feedSourceId: number | string) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (version: number) =>
      apiPost<unknown>(`/feed-sources/${feedSourceId}/export-history/${version}/rollback`, {}),
    onSuccess: () => {
      void queryClient.invalidateQueries({ queryKey: queryKeys.feedSource(feedSourceId).exportHistory });
    },
  });
}
```

```tsx
import { Accordion, Alert, Code, Group, Stack, Table, Text } from '@mantine/core';
import { useTranslation } from 'react-i18next';
import { EmptyState, ErrorState, LoadingState } from '../../components/StateViews';
import type { DiffOut } from '../../api/types';

type Props = {
  diff: DiffOut | undefined;
  isPending: boolean;
  isError: boolean;
  onRetry: () => void;
};

export function ExportVersionDiff({ diff, isPending, isError, onRetry }: Props) {
  const { t } = useTranslation('export');
  if (isPending) return <LoadingState />;
  if (isError) return <ErrorState onRetry={onRetry} />;
  if (!diff) return <EmptyState>{t('selectVersions')}</EmptyState>;
  const hasChanges = diff.added.length > 0 || diff.removed.length > 0 || diff.changed.length > 0;
  if (!hasChanges) return <EmptyState>{t('noChanges')}</EmptyState>;
  return (
    <Stack gap="md" data-testid="export-version-diff">
      <Text fw={600}>
        {t('diffTitle', { version: diff.version, against: diff.against })}
      </Text>
      {diff.added.length > 0 ? (
        <Alert color="green" title={t('added')}>
          <Code block>{diff.added.join(', ')}</Code>
        </Alert>
      ) : null}
      {diff.removed.length > 0 ? (
        <Alert color="red" title={t('removed')}>
          <Code block>{diff.removed.join(', ')}</Code>
        </Alert>
      ) : null}
      {diff.changed.length > 0 ? (
        <Accordion variant="separated" multiple>
          {diff.changed.map((product) => (
            <Accordion.Item key={product.product_id} value={product.product_id}>
              <Accordion.Control>
                <Group justify="space-between">
                  <Text fw={500}>{product.product_id}</Text>
                  <Text size="sm" c="dimmed">
                    {t('fieldsChanged', { count: product.fields.length })}
                  </Text>
                </Group>
              </Accordion.Control>
              <Accordion.Panel>
                <Table withTableBorder>
                  <Table.Thead>
                    <Table.Tr>
                      <Table.Th>{t('columns.field')}</Table.Th>
                      <Table.Th>{t('columns.old')}</Table.Th>
                      <Table.Th>{t('columns.new')}</Table.Th>
                    </Table.Tr>
                  </Table.Thead>
                  <Table.Tbody>
                    {product.fields.map((field, idx) => (
                      <Table.Tr key={`${field.field}-${idx}`}>
                        <Table.Td>{field.field}</Table.Td>
                        <Table.Td>
                          <Code>{JSON.stringify(field.old)}</Code>
                        </Table.Td>
                        <Table.Td>
                          <Code>{JSON.stringify(field.new)}</Code>
                        </Table.Td>
                      </Table.Tr>
                    ))}
                  </Table.Tbody>
```

```tsx
import { Stack, Title } from '@mantine/core';
import { useEffect, useMemo, useState } from 'react';
import { useParams } from 'react-router';
import { useTranslation } from 'react-i18next';
import { useExportHistory, useExportVersionDiff, useRollbackToVersion, useFeedSource } from '../../api/hooks';
import { ExportUrlBlock } from '../../components/ExportUrlBlock';
import { EmptyState, ErrorState, LoadingState } from '../../components/StateViews';
import { notifyMutationError, notifySuccess } from '../../app/notifications';
import { ApiError } from '../../api/client';
import { ExportVersionList } from './ExportVersionList';
import { ExportVersionDiff } from './ExportVersionDiff';
import { RollbackConfirmModal } from './RollbackConfirmModal';

export function ExportPage() {
  const { t } = useTranslation('export');
  const { feedSourceId } = useParams();
  const id = feedSourceId ?? '';
  const feed = useFeedSource(id);
  const history = useExportHistory(id);
  const rollback = useRollbackToVersion(id);
  const [versionA, setVersionA] = useState<number | undefined>();
  const [versionB, setVersionB] = useState<number | undefined>();
  const [rollbackTarget, setRollbackTarget] = useState<number | null>(null);

  const versions = history.data ?? [];

  useEffect(() => {
    if (versions.length >= 2 && versionA === undefined && versionB === undefined) {
      setVersionA(versions[0].version);
      setVersionB(versions[1].version);
    } else if (versions.length === 1 && versionA === undefined) {
      setVersionA(versions[0].version);
    }
  }, [versions, versionA, versionB]);

  const diff = useExportVersionDiff(id, versionA, versionB);

  if (feed.isPending || history.isPending) return <LoadingState />;
  if (feed.isError) return <ErrorState onRetry={() => void feed.refetch()} />;
  if (history.isError) return <ErrorState onRetry={() => void history.refetch()} />;
  if (!feed.data) return <EmptyState>{t('feedNotFound')}</EmptyState>;

  async function onConfirmRollback(version: number) {
    try {
      await rollback.mutateAsync(version);
      notifySuccess(t('rollbackSuccess', { version }));
      setRollbackTarget(null);
    } catch (error) {
      notifyMutationError(error, t('rollbackFailed'));
    }
  }

  return (
    <Stack gap="md">
      <Title order={3}>{t('title')}</Title>
      {feed.data.export_url ? (
        <ExportUrlBlock
          feedSourceId={id}
          exportUrl={feed.data.export_url}
          onRotated={() => void feed.refetch()}
        />
      ) : null}
      {versions.length === 0 ? (
        <EmptyState>{t('versions.empty')}</EmptyState>
      ) : (
        <>
          <ExportVersionList
            versions={versions}
            versionA={versionA}
            versionB={versionB}
            onSelectA={setVersionA}
            onSelectB={setVersionB}
            onRollback={setRollbackTarget}
          />
          <ExportVersionDiff
            diff={diff.data}
            isPending={diff.isPending}
            isError={diff.isError}
            onRetry={() => void diff.refetch()}
          />
        </>
      )}
      <RollbackConfirmModal
        opened={rollbackTarget !== null}
        version={rollbackTarget}
        onClose={() => setRollbackTarget(null)}
        onConfirm={onConfirmRollback}
        pending={rollback.isPending}
      />
    </Stack>
  );
}
```
