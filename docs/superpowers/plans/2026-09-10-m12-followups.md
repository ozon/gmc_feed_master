def test_index_rebuilds_on_same_file_edit(tmp_path: Path) -> None:
    """Writing a different CSV to the same path triggers a rebuild."""
    csv_a = "1,Food > Fruit\n2,Food > Veg\n"
    csv_b = "10,Home > Kitchen\n20,Home > Garden\n"
    p = tmp_path / "taxonomy.csv"
    p.write_text(csv_a, encoding="utf-8")
    idx = TaxonomyIndex(p, language="en")
    assert idx.contains("1")
    assert not idx.contains("10")
    # Overwrite same file (same path, no invalidate)
    p.write_text(csv_b, encoding="utf-8")
    # Next access triggers rebuild (mtime/size changed)
    assert not idx.contains("1")
    assert idx.contains("10")
```

# In plugins/core/category/plugin.py — add a new public function:
def collect_config_errors(config: Any) -> list[str]:
    """Return every validation error for a category config, not just the first."""
    errors: list[str] = []
    rules = config.get("rules") if isinstance(config, dict) else None
    if not isinstance(rules, list):
        return ["rules must be a non-empty array"]
    for i, rule in enumerate(rules):
        prefix = f"rules[{i}]"
        if not isinstance(rule, dict):
            errors.append(f"{prefix}: must be a dict")
            continue
        # source_field
        sf = rule.get("source_field")
        if not isinstance(sf, str) or not sf.strip():
            errors.append(f"{prefix}: source_field must be a non-empty string")
        # target_value
        tv = rule.get("target_value")
        if not isinstance(tv, str) or not tv.strip():
            errors.append(f"{prefix}: target_value must be a non-empty string")
        # is_excluded
        ie = rule.get("is_excluded")
        if not isinstance(ie, bool):
            errors.append(f"{prefix}: is_excluded must be a boolean")
        # in (optional list)
        in_list = rule.get("in")
        if in_list is not None:
            if not isinstance(in_list, list):
                errors.append(f"{prefix}: in must be a list")
            else:
                for j, entry in enumerate(in_list):
                    if not isinstance(entry, str):
                        errors.append(f"{prefix}: in[{j}] must be a string")
                    elif not entry.strip():
                        errors.append(f"{prefix}: in[{j}] must not be empty")
    return errors

# Refactor validate_config to reuse it:
def validate_config(config: Any) -> None:
    errors = collect_config_errors(config)
    if errors:
        raise ValueError("; ".join(errors))
```
